// --- ADVANCED AUDIO ENGINE ---
object LimboSoundEngine {
    var soundEnabled = true
    var volume = 0.5f
    var soundType = "Cyber Bell"

    fun playSound(pitchMultiplier: Float = 1.0f) {
        if (!soundEnabled || volume <= 0.01f) return
        thread {
            try {
                val sampleRate = 22050
                val durationMs = when (soundType) {
                    "Digital Ring" -> 100
                    "Glitch Beep" -> 50
                    else -> 140
                }
                val numSamples = durationMs * sampleRate / 1000
                val buffer = ShortArray(numSamples)

                val baseFreq1 = when (soundType) {
                    "Digital Ring" -> 1046.50f
                    "Glitch Beep" -> 520.0f
                    else -> 880.0f
                }
                val baseFreq2 = if (soundType == "Digital Ring") 1318.51f else 1174.66f

                val f1 = baseFreq1 * pitchMultiplier
                val f2 = baseFreq2 * pitchMultiplier

                for (i in 0 until numSamples) {
                    val t = i.toDouble() / sampleRate
                    val envelope = Math.exp(-t * (if (soundType == "Digital Ring") 28.0 else 18.0))
                    
                    val wave1 = if (soundType == "Glitch Beep") {
                        if ((t * 120).toInt() % 2 == 0) 0.7 else -0.7
                    } else {
                        Math.sin(2.0 * Math.PI * f1 * t)
                    }
                    val wave2 = Math.sin(2.0 * Math.PI * f2 * t) * 0.4
                    val ringMod = if (soundType == "Digital Ring") Math.sin(2.0 * Math.PI * 40.0 * t) else 1.0

                    val sampleValue = ((wave1 + wave2) * 0.5 * ringMod * envelope * Short.MAX_VALUE * volume).toInt()
                    buffer[i] = sampleValue.coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort()
                }

                val track = AudioTrack.Builder()
                    .setAudioAttributes(
                        AudioAttributes.Builder()
                            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
                            .setUsage(AudioAttributes.USAGE_ASSISTANCE_SONIFICATION)
                            .build()
                    )
                    .setAudioFormat(
                        AudioFormat.Builder()
                            .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                            .setSampleRate(sampleRate)
                            .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                            .build()
                    )
                    .setBufferSizeInBytes(buffer.size * 2)
                    .setTransferMode(AudioTrack.MODE_STATIC)
                    .build()

                track.write(buffer, 0, buffer.size)
                track.play()
                Thread.sleep(durationMs.toLong())
                track.release()
            } catch (_: Exception) {}
        }
    }
}

@OptIn(ExperimentalFoundationApi::class)
@Composable
fun LimboLauncherAdvanced() {
    val context = LocalContext.current
    val prefs = remember { context.getSharedPreferences("limbo_config", Context.MODE_PRIVATE) }

    var bgColorHex by remember { mutableStateOf(prefs.getLong("bg_color", Color.Black.value.toLong())) }
    var accentColorHex by remember { mutableStateOf(prefs.getLong("accent_color", Color.White.value.toLong())) }
    var iconStyle by remember { mutableStateOf(prefs.getString("icon_style", "Monochrome") ?: "Monochrome") }
    var bgAnimation by remember { mutableStateOf(prefs.getString("bg_anim", "✦") ?: "✦") }
    var soundActive by remember { mutableStateOf(prefs.getBoolean("sound_active", true)) }
    var soundVolume by remember { mutableFloatStateOf(prefs.getFloat("sound_volume", 0.5f)) }
    var selectedSoundType by remember { mutableStateOf(prefs.getString("sound_type", "Cyber Bell") ?: "Cyber Bell") }

    LimboSoundEngine.soundEnabled = soundActive
    LimboSoundEngine.volume = soundVolume
    LimboSoundEngine.soundType = selectedSoundType

    val bgColor = Color(bgColorHex.toULong())
    val accentColor = Color(accentColorHex.toULong())

    fun saveConfig() {
        prefs.edit()
            .putLong("bg_color", bgColorHex)
            .putLong("accent_color", accentColorHex)
            .putString("icon_style", iconStyle)
            .putString("bg_anim", bgAnimation)
            .putBoolean("sound_active", soundActive)
            .putFloat("sound_volume", soundVolume)
            .putString("sound_type", selectedSoundType)
            .apply()
    }

    var searchQuery by remember { mutableStateOf("") }
    var showAppDrawer by remember { mutableStateOf(false) }
    var showConfigMenu by remember { mutableStateOf(false) }
    var showSysMonitor by remember { mutableStateOf(false) }

    var selectedAppContextMenu by remember { mutableStateOf<ResolveInfo?>(null) }

    BackHandler(enabled = selectedAppContextMenu != null || showAppDrawer || showConfigMenu || showSysMonitor) {
        LimboSoundEngine.playSound(0.8f)
        when {
            selectedAppContextMenu != null -> selectedAppContextMenu = null
            showSysMonitor -> showSysMonitor = false
            showConfigMenu -> showConfigMenu = false
            showAppDrawer -> showAppDrawer = false
        }
    }

    val installedApps = remember {
        val mainIntent = Intent(Intent.ACTION_MAIN, null).apply {
            addCategory(Intent.CATEGORY_LAUNCHER)
        }
        val pkgManager = context.packageManager
        pkgManager.queryIntentActivities(mainIntent, 0).sortedBy {
            it.loadLabel(pkgManager).toString().lowercase()
        }
    }

    val pinnedPackageNames = remember {
        val savedSet = prefs.getStringSet("pinned_apps", emptySet()) ?: emptySet()
        mutableStateListOf(*savedSet.toTypedArray())
    }

    fun savePinnedApps() {
        prefs.edit().putStringSet("pinned_apps", pinnedPackageNames.toSet()).apply()
    }

    val pinnedApps = installedApps.filter { pinnedPackageNames.contains(it.activityInfo.packageName) }
    val filteredApps = installedApps.filter {
        val appName = it.loadLabel(context.packageManager).toString()
        appName.contains(searchQuery, ignoreCase = true)
    }

    Surface(
        modifier = Modifier.fillMaxSize(),
        color = bgColor
    ) {
        Box(modifier = Modifier.fillMaxSize()) {
            Row(modifier = Modifier.fillMaxSize()) {

                // DOCK / SEITENLEISTE
                Column(
                    modifier = Modifier
                        .width(70.dp)
                        .fillMaxHeight()
                        .padding(vertical = 8.dp, horizontal = 4.dp)
                        .background(bgColor, RectangleShape)
                        .border(1.dp, accentColor, RectangleShape),
                    horizontalAlignment = Alignment.CenterHorizontally,
                    verticalArrangement = Arrangement.Top
                ) {
                    Spacer(modifier = Modifier.height(8.dp))

                    Box(
                        modifier = Modifier
                            .size(48.dp)
                            .background(bgColor, RectangleShape)
                            .border(1.dp, accentColor, RectangleShape)
                            .clickable {
                                LimboSoundEngine.playSound(1.2f)
                                showConfigMenu = false
                                showSysMonitor = false
                                selectedAppContextMenu = null
                                showAppDrawer = !showAppDrawer
                            },
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            text = "[::]",
                            color = accentColor,
                            fontSize = 18.sp,
                            fontWeight = FontWeight.Bold
                        )
                    }

                    Spacer(modifier = Modifier.height(8.dp))
                    Box(
                        modifier = Modifier
                            .width(40.dp)
                            .height(1.dp)
                            .background(accentColor)
                    )
                    Spacer(modifier = Modifier.height(8.dp))

                    Column(
                        verticalArrangement = Arrangement.spacedBy(8.dp),
                        horizontalAlignment = Alignment.CenterHorizontally,
                        modifier = Modifier.weight(1f)
                    ) {
                        pinnedApps.forEach { app ->
                            AppIconView(
                                appInfo = app,
                                iconStyle = iconStyle,
                                bgColor = bgColor,
                                accentColor = accentColor,
                                onClick = {
                                    LimboSoundEngine.playSound(1.4f)
                                    val launchIntent = context.packageManager.getLaunchIntentForPackage(app.activityInfo.packageName)
                                    launchIntent?.let { context.startActivity(it) }
                                },
                                onLongClick = {
                                    LimboSoundEngine.playSound(0.7f)
                                    selectedAppContextMenu = app
                                }
                            )
                        }
                    }
                }

                // HAUPTBEREICH
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                        .padding(12.dp)
                ) {
                    when {
                        showSysMonitor -> {
                            SystemMonitorOverlay(
                                accentColor = accentColor,
                                bgColor = bgColor,
                                onClose = {
                                    LimboSoundEngine.playSound(0.8f)
                                    showSysMonitor = false
                                }
                            )
                        }
                        showConfigMenu -> {
                            LimboConfigScreen(
                                bgColor = bgColor,
                                accentColor = accentColor,
                                iconStyle = iconStyle,
                                bgAnimation = bgAnimation,
                                soundActive = soundActive,
                                soundVolume = soundVolume,
                                selectedSoundType = selectedSoundType,
                                onBgColorChange = {
                                    LimboSoundEngine.playSound(1.0f)
                                    bgColorHex = it.value.toLong()
                                    saveConfig()
                                },
                                onAccentColorChange = {
                                    LimboSoundEngine.playSound(1.1f)
                                    accentColorHex = it.value.toLong()
                                    saveConfig()
                                },
                                onIconStyleChange = {
                                    LimboSoundEngine.playSound(1.0f)
                                    iconStyle = it
                                    saveConfig()
                                },
                                onBgAnimChange = {
                                    LimboSoundEngine.playSound(1.05f)
                                    bgAnimation = it
                                    saveConfig()
                                },
                                onSoundToggle = {
                                    soundActive = !soundActive
                                    LimboSoundEngine.soundEnabled = soundActive
                                    LimboSoundEngine.playSound(1.2f)
                                    saveConfig()
                                },
                                onVolumeChange = {
                                    soundVolume = it
                                    LimboSoundEngine.volume = it
                                    saveConfig()
                                },
                                onSoundTypeChange = {
                                    selectedSoundType = it
                                    LimboSoundEngine.soundType = it
                                    LimboSoundEngine.playSound(1.0f)
                                    saveConfig()
                                },
                                onClose = {
                                    LimboSoundEngine.playSound(0.8f)
                                    showConfigMenu = false
                                }
                            )
                        }
                        showAppDrawer -> {
                            Column(modifier = Modifier.fillMaxSize()) {

                                OutlinedTextField(
                                    value = searchQuery,
                                    onValueChange = {
                                        searchQuery = it
                                        LimboSoundEngine.playSound(0.9f)
                                        val queryClean = it.trim().lowercase()
                                        if (queryClean == "limbo config" || queryClean == "config") {
                                            LimboSoundEngine.playSound(1.3f)
                                            showConfigMenu = true
                                        } else if (queryClean == "top" || queryClean == "sys") {
                                            LimboSoundEngine.playSound(1.3f)
                                            showSysMonitor = true
                                        }
                                    },
                                    modifier = Modifier.fillMaxWidth(),
                                    placeholder = { Text("App suchen...", color = accentColor.copy(alpha = 0.5f)) },
                                    singleLine = true,
                                    shape = RectangleShape,
                                    colors = OutlinedTextFieldDefaults.colors(
                                        focusedBorderColor = accentColor,
                                        unfocusedBorderColor = accentColor,
                                        focusedTextColor = accentColor,
                                        unfocusedTextColor = accentColor,
                                        cursorColor = accentColor
                                    )
                                )

                                Spacer(modifier = Modifier.height(12.dp))

                                LazyVerticalGrid(
                                    columns = GridCells.Adaptive(minSize = 75.dp),
                                    horizontalArrangement = Arrangement.spacedBy(8.dp),
                                    verticalArrangement = Arrangement.spacedBy(12.dp),
                                    modifier = Modifier.fillMaxSize()
                                ) {
                                    items(filteredApps) { appInfo ->
                                        MonochromeAppItem(
                                            appInfo = appInfo,
                                            iconStyle = iconStyle,
                                            bgColor = bgColor,
                                            accentColor = accentColor,
                                            onClick = {
                                                LimboSoundEngine.playSound(1.4f)
                                                val launchIntent = context.packageManager.getLaunchIntentForPackage(appInfo.activityInfo.packageName)
                                                launchIntent?.let { context.startActivity(it) }
                                            },
                                            onLongClick = {
                                                LimboSoundEngine.playSound(0.9f)
                                                selectedAppContextMenu = appInfo
                                            }
                                        )
                                    }
                                }
                            }
                        }
                        else -> {
                            // HOMESCREEN
                            Box(
                                modifier = Modifier
                                    .fillMaxSize()
                                    .combinedClickable(
                                        onClick = {
                                            // Kurzer Klick = Sys Monitor
                                            LimboSoundEngine.playSound(1.5f)
                                            showSysMonitor = true
                                        },
                                        onLongClick = {
                                            // Langer Klick = Config-Menü
                                            LimboSoundEngine.playSound(1.3f)
                                            showConfigMenu = true
                                        }
                                    ),
                                contentAlignment = Alignment.Center
                            ) {
                                AnimatedBackground(symbol = bgAnimation, accentColor = accentColor)

                                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                                    Text(
                                        text = "LIMBO OS V9.1",
                                        color = accentColor,
                                        fontSize = 32.sp,
                                        fontWeight = FontWeight.Bold,
                                        letterSpacing = 4.sp
                                    )
                                    Spacer(modifier = Modifier.height(8.dp))
                                    Text(
                                        text = "[ Halten für Config | Klick für Sys ]",
                                        color = accentColor.copy(alpha = 0.5f),
                                        fontSize = 11.sp,
                                        fontFamily = FontFamily.Monospace
                                    )
                                }
                            }
                        }
                    }
                }
            }

            // APP KONTEXT-MENÜ POPUP
            selectedAppContextMenu?.let { app ->
                val pkgName = app.activityInfo.packageName
                val appLabel = app.loadLabel(context.packageManager).toString()
                val isPinned = pinnedPackageNames.contains(pkgName)

                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(Color.Black.copy(alpha = 0.7f))
                        .clickable {
                            LimboSoundEngine.playSound(0.7f)
                            selectedAppContextMenu = null
                        },
                    contentAlignment = Alignment.Center
                ) {
                    Column(
                        modifier = Modifier
                            .width(260.dp)
                            .background(bgColor)
                            .border(1.dp, accentColor, RectangleShape)
                            .padding(16.dp)
                            .clickable(enabled = false) {},
                        horizontalAlignment = Alignment.CenterHorizontally
                    ) {
                        Text(
                            text = appLabel.uppercase(),
                            color = accentColor,
                            fontSize = 14.sp,
                            fontWeight = FontWeight.Bold,
                            textAlign = TextAlign.Center,
                            fontFamily = FontFamily.Monospace,
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis
                        )

                        Spacer(modifier = Modifier.height(12.dp))
                        Box(modifier = Modifier.fillMaxWidth().height(1.dp).background(accentColor))
                        Spacer(modifier = Modifier.height(12.dp))

                        Button(
                            onClick = {
                                LimboSoundEngine.playSound(1.2f)
                                if (isPinned) {
                                    pinnedPackageNames.remove(pkgName)
                                } else {
                                    pinnedPackageNames.add(pkgName)
                                }
                                savePinnedApps()
                                selectedAppContextMenu = null
                            },
                            shape = RectangleShape,
                            colors = ButtonDefaults.buttonColors(containerColor = bgColor, contentColor = accentColor),
                            modifier = Modifier.fillMaxWidth().border(1.dp, accentColor, RectangleShape)
                        ) {
                            Text(if (isPinned) "Aus Leiste entfernen" else "In Leiste pinnen", fontSize = 12.sp)
                        }

                        Spacer(modifier = Modifier.height(8.dp))

                        Button(
                            onClick = {
                                LimboSoundEngine.playSound(1.1f)
                                selectedAppContextMenu = null
                                val intent = Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
                                    data = Uri.fromParts("package", pkgName, null)
                                }
                                context.startActivity(intent)
                            },
                            shape = RectangleShape,
                            colors = ButtonDefaults.buttonColors(containerColor = bgColor, contentColor = accentColor),
                            modifier = Modifier.fillMaxWidth().border(1.dp, accentColor, RectangleShape)
                        ) {
                            Text("App-Informationen", fontSize = 12.sp)
                        }

                        Spacer(modifier = Modifier.height(8.dp))

                        Button(
                            onClick = {
                                LimboSoundEngine.playSound(0.6f)
                                selectedAppContextMenu = null
                                val intent = Intent(Intent.ACTION_DELETE).apply {
                                    data = Uri.parse("package:$pkgName")
                                }
                                context.startActivity(intent)
                            },
                            shape = RectangleShape,

// --- SYSTEM MONITOR DASHBOARD ---
@Composable
fun SystemMonitorOverlay(
    accentColor: Color,
    bgColor: Color,
    onClose: () -> Unit
) {
    val context = LocalContext.current
    var memoryUsage by remember { mutableStateOf("Berechne...") }
    var batteryLevel by remember { mutableIntStateOf(0) }
    var uptimeString by remember { mutableStateOf("") }

    LaunchedEffect(Unit) {
        val runtime = Runtime.getRuntime()
        val usedMemMB = (runtime.totalMemory() - runtime.freeMemory()) / (1024 * 1024)
        val maxMemMB = runtime.maxMemory() / (1024 * 1024)
        memoryUsage = "$usedMemMB MB / $maxMemMB MB"

        val batteryIntent = context.registerReceiver(null, IntentFilter(Intent.ACTION_BATTERY_CHANGED))
        val level = batteryIntent?.getIntExtra(BatteryManager.EXTRA_LEVEL, -1) ?: -1
        val scale = batteryIntent?.getIntExtra(BatteryManager.EXTRA_SCALE, -1) ?: -1
        batteryLevel = if (level != -1 && scale != -1) ((level.toFloat() / scale.toFloat()) * 100).toInt() else 0

        val uptimeSec = SystemClock.elapsedRealtime() / 1000
        val hours = uptimeSec / 3600
        val mins = (uptimeSec % 3600) / 60
        uptimeString = "${hours}h ${mins}m"
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .border(1.dp, accentColor, RectangleShape)
            .background(bgColor)
            .padding(16.dp)
    ) {
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(
                text = "SYSTEM STATUS [DEBUG]",
                color = accentColor,
                fontSize = 16.sp,
                fontWeight = FontWeight.Bold,
                fontFamily = FontFamily.Monospace
            )
            Button(
                onClick = onClose,
                shape = RectangleShape,
                colors = ButtonDefaults.buttonColors(containerColor = accentColor, contentColor = bgColor)
            ) {
                Text("X")
            }
        }

        Spacer(modifier = Modifier.height(16.dp))

        Text("• HEAP RAM: $memoryUsage", color = accentColor, fontFamily = FontFamily.Monospace, fontSize = 14.sp)
        Spacer(modifier = Modifier.height(8.dp))
        Text("• BATTERY: $batteryLevel%", color = accentColor, fontFamily = FontFamily.Monospace, fontSize = 14.sp)
        Spacer(modifier = Modifier.height(8.dp))
        Text("• UPTIME: $uptimeString", color = accentColor, fontFamily = FontFamily.Monospace, fontSize = 14.sp)
        Spacer(modifier = Modifier.height(8.dp))
        Text("• CPU THREADS: ${Runtime.getRuntime().availableProcessors()}", color = accentColor, fontFamily = FontFamily.Monospace, fontSize = 14.sp)
        Spacer(modifier = Modifier.height(8.dp))
        Text("• PID: ${Process.myPid()}", color = accentColor, fontFamily = FontFamily.Monospace, fontSize = 14.sp)

        Spacer(modifier = Modifier.height(24.dp))
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(1.dp)
                .background(accentColor)
        )
        Spacer(modifier = Modifier.height(16.dp))

        Text(
            text = "KERNEL STATUS: OK\nSHELL: ARCH/LINUX EMULATION\nREADY FOR TERMINAL COMMANDS...",
            color = accentColor.copy(alpha = 0.7f),
            fontFamily = FontFamily.Monospace,
            fontSize = 12.sp
        )
    }
}

// --- ENGINE FÜR ANIMIERTE HINTERGRÜNDE ---
@Composable
fun AnimatedBackground(symbol: String, accentColor: Color) {
    if (symbol == "✕") return

    val infiniteTransition = rememberInfiniteTransition(label = "bg_engine")
    val animVal by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 1f,
        animationSpec = infiniteRepeatable(
            animation = tween(3500, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "loop"
    )

    Canvas(modifier = Modifier.fillMaxSize()) {
        when (symbol) {
            "✦" -> {
                val stars = List(50) { Offset(Random(it).nextFloat(), Random(it * 2).nextFloat()) to (Random(it * 3).nextFloat() * 2.5f + 1f) }
                stars.forEach { (pos, radius) ->
                    val alpha = (sin((animVal * 6.28f) + pos.x * 10f) + 1f) / 2f
                    drawCircle(color = accentColor.copy(alpha = alpha * 0.7f), radius = radius.dp.toPx(), center = Offset(pos.x * size.width, pos.y * size.height))
                }
            }
            "░" -> {
                val drops = List(30) { Random(it).nextFloat() }
                drops.forEachIndexed { i, startX ->
                    val y = ((animVal + i * 0.08f) % 1f) * size.height
                    drawCircle(color = accentColor.copy(alpha = 0.8f), radius = 2.dp.toPx(), center = Offset(startX * size.width, y))
                }
            }
            "⬡" -> {
                val radius = 30.dp.toPx()
                val hexWidth = radius * 1.732f
                val hexHeight = radius * 2f
                var y = 0f
                var row = 0
                while (y < size.height + hexHeight) {
                    var x = if (row % 2 == 0) 0f else hexWidth / 2f
                    while (x < size.width + hexWidth) {
                        val pulse = (sin(animVal * 6.28f + (x + y) * 0.01f) + 1f) / 2f
                        drawCircle(color = accentColor.copy(alpha = pulse * 0.2f), radius = 2.dp.toPx(), center = Offset(x, y))
                        x += hexWidth
                    }
                    y += hexHeight * 0.75f
                    row++
                }
            }
            "═" -> {
                val step = 50.dp.toPx()
                val offsetAnim = animVal * step
                var y = offsetAnim
                while (y < size.height) {
                    drawLine(color = accentColor.copy(alpha = 0.15f), start = Offset(0f, y), end = Offset(size.width, y), strokeWidth = 1.dp.toPx())
                    y += step
                }
            }
            "◌" -> {
                val center = Offset(size.width / 2f, size.height / 2f)
                val maxRadius = size.width.coerceAtLeast(size.height) * 0.7f
                for (i in 0..2) {
                    val r = ((animVal + i * 0.33f) % 1f) * maxRadius
                    val alpha = (1f - (r / maxRadius)).coerceIn(0f, 1f)
                    drawCircle(color = accentColor.copy(alpha = alpha * 0.3f), radius = r, center = center, style = androidx.compose.ui.graphics.drawscope.Stroke(width = 1.5.dp.toPx()))
                }
            }
            "∿" -> {
                val points = 30
                val dx = size.width / points
                for (i in 0 until points) {
                    val x1 = i * dx
                    val y1 = size.height / 2f + sin(animVal * 6.28f + i * 0.3f) * 40.dp.toPx()
                    val x2 = (i + 1) * dx
                    val y2 = size.height / 2f + sin(animVal * 6.28f + (i + 1) * 0.3f) * 40.dp.toPx()
                    drawLine(color = accentColor.copy(alpha = 0.4f), start = Offset(x1, y1), end = Offset(x2, y2), strokeWidth = 2.dp.toPx())
                }
            }
            "𐓷" -> {
                val center = Offset(size.width / 2f, size.height / 2f)
                val particles = List(40) { Random(it).nextFloat() to Random(it * 3).nextFloat() * 6.28f }
                particles.forEach { (speed, angle) ->
                    val dist = ((animVal * speed) % 1f) * size.width * 0.6f
                    val x = center.x + cos(angle) * dist
                    val y = center.y + sin(angle) * dist
                    drawCircle(color = accentColor.copy(alpha = (dist / (size.width * 0.6f)) * 0.7f), radius = 2.dp.toPx(), center = Offset(x, y))
                }
            }
            "❖" -> {
                val center = Offset(size.width / 2f, size.height / 2f)
                for (i in 1..4) {
                    val sz = (i * 40.dp.toPx() + animVal * 30.dp.toPx()) % (size.width * 0.5f)
                    val alpha = 1f - (sz / (size.width * 0.5f))
                    drawLine(accentColor.copy(alpha = alpha * 0.3f), Offset(center.x - sz, center.y), Offset(center.x, center.y - sz), 2.dp.toPx())
                    drawLine(accentColor.copy(alpha = alpha * 0.3f), Offset(center.x, center.y - sz), Offset(center.x + sz, center.y), 2.dp.toPx())
                    drawLine(accentColor.copy(alpha = alpha * 0.3f), Offset(center.x + sz, center.y), Offset(center.x, center.y + sz), 2.dp.toPx())
                    drawLine(accentColor.copy(alpha = alpha * 0.3f), Offset(center.x, center.y + sz), Offset(center.x - sz, center.y), 2.dp.toPx())
                }
            }
            "▲" -> {

                    val y = (size.height - ((animVal * size.height + i * 50f) % size.height))

                }

            "✦═" -> {

            }
            "🌀" -> {
                val center = Offset(size.width / 2f, size.height / 2f)
                for (i in 0..30) {
                    val angle = animVal * 6.28f + (i * 0.2f)

                }
            }
            "❄" -> {

                    drawLine(accentColor.copy(alpha = alpha), Offset(center.x - side, center.y - side), Offset(center.x + side, center.y - side), strokeWidth = 1.dp.toPx())
                    drawLine(accentColor.copy(alpha = alpha), Offset(center.x + side, center.y - side), Offset(center.x + side, center.y + side), strokeWidth = 1.dp.toPx())
                    drawLine(accentColor.copy(alpha = alpha), Offset(center.x + side, center.y + side), Offset(center.x - side, center.y + side), strokeWidth = 1.dp.toPx())

// --- APP ICON COMPONENT ---
@OptIn(ExperimentalFoundationApi::class)
@Composable
fun AppIconView(
    appInfo: ResolveInfo,
    iconStyle: String,
    bgColor: Color,
    accentColor: Color,
    onClick: () -> Unit,
    onLongClick: () -> Unit
) {
    val context = LocalContext.current
    val appName = remember(appInfo) { appInfo.loadLabel(context.packageManager).toString() }

    Box(
        modifier = Modifier
            .size(48.dp)
            .background(bgColor, RectangleShape)
            .border(1.dp, accentColor, RectangleShape)
            .combinedClickable(onClick = onClick, onLongClick = onLongClick),
        contentAlignment = Alignment.Center
    ) {
        if (iconStyle == "System Icons") {
            var iconBitmap by remember(appInfo) { mutableStateOf<androidx.compose.ui.graphics.ImageBitmap?>(null) }

            LaunchedEffect(appInfo) {
                try {
                    val drawable = appInfo.loadIcon(context.packageManager)
                    iconBitmap = drawable.toBitmap().asImageBitmap()
                } catch (_: Exception) {
                    iconBitmap = null
                }
            }

            if (iconBitmap != null) {
                Image(
                    bitmap = iconBitmap!!,
                    contentDescription = appName,
                    modifier = Modifier.size(34.dp)
                )
            } else {
                Text(
                    text = nameToInitials(appName),
                    color = accentColor,
                    fontWeight = FontWeight.Bold,
                    fontSize = 18.sp
                )
            }
        } else {
            Text(
                text = nameToInitials(appName),
                color = accentColor,
                fontWeight = FontWeight.Bold,
                fontSize = 18.sp
            )
        }
    }
}

@OptIn(ExperimentalFoundationApi::class)
@Composable
fun MonochromeAppItem(
    appInfo: ResolveInfo,
    iconStyle: String,
    bgColor: Color,
    accentColor: Color,
    onClick: () -> Unit,
    onLongClick: () -> Unit
) {
    val context = LocalContext.current
    val appName = remember(appInfo) { appInfo.loadLabel(context.packageManager).toString() }

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        modifier = Modifier.padding(4.dp)
    ) {
        AppIconView(
            appInfo = appInfo,
            iconStyle = iconStyle,
            bgColor = bgColor,
            accentColor = accentColor,
            onClick = onClick,
            onLongClick = onLongClick
        )
        Spacer(modifier = Modifier.height(4.dp))
        Text(
            text = appName,
            color = accentColor,
            fontSize = 10.sp,
            maxLines = 1,
            overflow = TextOverflow.Ellipsis,
            textAlign = TextAlign.Center
        )
    }
}

fun nameToInitials(name: String): String {
    return name.take(1).uppercase()
}
