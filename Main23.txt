// AUDIOSYNTHESIZER FÜR TERMINAL BEEP
object TerminalAudioSynth {
    fun playBeep(freqHz: Float = 880f, durationMs: Int = 40) {
        Thread {
            try {
                val sampleRate = 22050
                val numSamples = sampleRate * durationMs / 1000
                val sample = DoubleArray(numSamples)
                val generatedSnd = ByteArray(2 * numSamples)

                for (i in 0 until numSamples) {
                    sample[i] = sin(2 * Math.PI * i / (sampleRate / freqHz))
                }

                var idx = 0
                for (dVal in sample) {
                    val valInt = (dVal * 32767).toInt().toShort()
                    generatedSnd[idx++] = (valInt.toInt() and 0x00ff).toByte()
                    generatedSnd[idx++] = (valInt.toInt() and 0xff00 shr 8).toByte()
                }

                val audioTrack = AudioTrack.Builder()
                    .setAudioAttributes(
                        AudioAttributes.Builder()
                            .setUsage(AudioAttributes.USAGE_ASSISTANCE_SONIFICATION)
                            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
                            .build()
                    )
                    .setAudioFormat(
                        AudioFormat.Builder()
                            .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                            .setSampleRate(sampleRate)
                            .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                            .build()
                    )
                    .setBufferSizeInBytes(generatedSnd.size)
                    .build()

                audioTrack.write(generatedSnd, 0, generatedSnd.size)
                audioTrack.play()
                Thread.sleep(durationMs.toLong())
                audioTrack.release()
            } catch (_: Exception) {}
        }.start()
    }
}

@OptIn(ExperimentalFoundationApi::class)
@Composable
fun LimboLauncherAdvanced() {
    val context = LocalContext.current
    val prefs = remember { context.getSharedPreferences("limbo_config", Context.MODE_PRIVATE) }
    val coroutineScope = rememberCoroutineScope()

    var refreshTrigger by remember { mutableIntStateOf(0) }
    var isGlitching by remember { mutableStateOf(false) }
    var isHackerModeActive by remember { mutableStateOf(false) }

    var bgColorHex by remember { mutableStateOf(prefs.getLong("bg_color", Color.Black.value.toLong())) }
    var bgSecondaryColorHex by remember { mutableStateOf(prefs.getLong("bg_sec_color", Color.Black.value.toLong())) }
    var accentColorHex by remember { mutableStateOf(prefs.getLong("accent_color", Color(0xFF00F0FF).value.toLong())) }
    var accentSecondaryColorHex by remember { mutableStateOf(prefs.getLong("accent_sec_color", Color(0xFF8A00FF).value.toLong())) }

    var iconStyle by remember { mutableStateOf(prefs.getString("icon_style", "Monochrome") ?: "Monochrome") }
    var bgStyle by remember { mutableStateOf(prefs.getString("bg_style", "Starfield") ?: "Starfield") }
    var terminalEnabled by remember { mutableStateOf(prefs.getBoolean("terminal_enabled", true)) }
    var audioFeedbackEnabled by remember { mutableStateOf(prefs.getBoolean("audio_enabled", true)) }

    val baseBgColor = Color(bgColorHex.toULong())
    val baseBgSecColor = Color(bgSecondaryColorHex.toULong())
    val baseAccentColor = Color(accentColorHex.toULong())
    val baseAccentSecColor = Color(accentSecondaryColorHex.toULong())

    val bgColor = if (isHackerModeActive) Color(0xFF100000) else baseBgColor
    val bgSecColor = if (isHackerModeActive) Color(0xFF2A0000) else baseBgSecColor
    val accentColor = if (isHackerModeActive) Color(0xFFFF0033) else baseAccentColor
    val accentSecColor = if (isHackerModeActive) Color(0xFFFF5500) else baseAccentSecColor

    val bgGradient = Brush.verticalGradient(colors = listOf(bgColor, bgSecColor))
    val accentGradient = Brush.verticalGradient(colors = listOf(accentColor, accentSecColor))

    fun triggerRmRfGlitch() {
        coroutineScope.launch {
            if (audioFeedbackEnabled) TerminalAudioSynth.playBeep(220f, 300)
            isGlitching = true
            delay(5000)
            isGlitching = false
        }
    }

    fun triggerLimboHackerMode() {
        coroutineScope.launch {
            if (audioFeedbackEnabled) TerminalAudioSynth.playBeep(1200f, 150)
            isHackerModeActive = true
            delay(30000)
            isHackerModeActive = false
        }
    }

    fun saveConfig() {
        prefs.edit()
            .putLong("bg_color", bgColorHex)
            .putLong("bg_sec_color", bgSecondaryColorHex)
            .putLong("accent_color", accentColorHex)
            .putLong("accent_sec_color", accentSecondaryColorHex)
            .putString("icon_style", iconStyle)
            .putString("bg_style", bgStyle)
            .putBoolean("terminal_enabled", terminalEnabled)
            .putBoolean("audio_enabled", audioFeedbackEnabled)
            .apply()
    }

    var searchQuery by remember { mutableStateOf("") }
    var showAppDrawer by remember { mutableStateOf(false) }
    var showConfigMenu by remember { mutableStateOf(false) }
    var showSysMonitor by remember { mutableStateOf(false) }
    var showTerminalWindow by remember { mutableStateOf(false) }
    var showHexViewerWindow by remember { mutableStateOf(false) }

    var selectedAppContextMenu by remember { mutableStateOf<ResolveInfo?>(null) }

    BackHandler(enabled = selectedAppContextMenu != null || showAppDrawer || showConfigMenu || showSysMonitor || showTerminalWindow || showHexViewerWindow) {
        when {
            selectedAppContextMenu != null -> selectedAppContextMenu = null
            showHexViewerWindow -> showHexViewerWindow = false
            showTerminalWindow -> showTerminalWindow = false
            showSysMonitor -> showSysMonitor = false
            showConfigMenu -> showConfigMenu = false
            showAppDrawer -> showAppDrawer = false
        }
    }

    val installedApps = remember(refreshTrigger) {
        val mainIntent = Intent(Intent.ACTION_MAIN, null).apply {
            addCategory(Intent.CATEGORY_LAUNCHER)
        }
        val pkgManager = context.packageManager
        pkgManager.queryIntentActivities(mainIntent, 0).sortedBy {
            it.loadLabel(pkgManager).toString().lowercase()
        }
    }

    val pinnedPackageNames = remember(refreshTrigger) {
        val savedSet = prefs.getStringSet("pinned_apps", emptySet()) ?: emptySet()
        mutableStateListOf(*savedSet.toTypedArray())
    }

    fun savePinnedApps() {
        prefs.edit().putStringSet("pinned_apps", pinnedPackageNames.toSet()).apply()
    }

    val pinnedApps = installedApps.filter { pinnedPackageNames.contains(it.activityInfo.packageName) }
    val filteredApps = installedApps.filter {
        val appName = it.loadLabel(context.packageManager).toString()
        appName.contains(searchQuery, ignoreCase = true)
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(bgGradient)
    ) {
        Row(modifier = Modifier.fillMaxSize()) {

            // DOCK / SEITENLEISTE
            Column(
                modifier = Modifier
                    .width(70.dp)
                    .fillMaxHeight()
                    .padding(vertical = 8.dp, horizontal = 4.dp)
                    .border(1.dp, accentGradient, RectangleShape),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.Top
            ) {
                Spacer(modifier = Modifier.height(8.dp))

                Box(
                    modifier = Modifier
                        .size(48.dp)
                        .border(1.dp, accentGradient, RectangleShape)
                        .clickable {
                            if (audioFeedbackEnabled) TerminalAudioSynth.playBeep(600f, 30)
                            showConfigMenu = false
                            showSysMonitor = false
                            showHexViewerWindow = false
                            selectedAppContextMenu = null
                            showAppDrawer = !showAppDrawer
                        },
                    contentAlignment = Alignment.Center
                ) {
                    Text(
                        text = "[::]",
                        style = TextStyle(brush = accentGradient),
                        fontSize = 18.sp,
                        fontWeight = FontWeight.Bold
                    )
                }

                if (terminalEnabled) {
                    Spacer(modifier = Modifier.height(8.dp))
                    Box(
                        modifier = Modifier
                            .size(48.dp)
                            .border(1.dp, accentGradient, RectangleShape)
                            .clickable {
                                if (audioFeedbackEnabled) TerminalAudioSynth.playBeep(880f, 30)
                                showTerminalWindow = !showTerminalWindow
                            },
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            text = "[>_]",
                            style = TextStyle(brush = accentGradient),
                            fontSize = 16.sp,
                            fontWeight = FontWeight.Bold
                        )
                    }
                }

                Spacer(modifier = Modifier.height(8.dp))
                // HEX VIEWER BUTTON FOR LINUX NERDS
                Box(
                    modifier = Modifier
                        .size(48.dp)
                        .border(1.dp, accentGradient, RectangleShape)
                        .clickable {
                            if (audioFeedbackEnabled) TerminalAudioSynth.playBeep(1000f, 30)
                            showHexViewerWindow = !showHexViewerWindow
                        },
                    contentAlignment = Alignment.Center
                ) {
                    Text(
                        text = "[0x]",
                        style = TextStyle(brush = accentGradient),
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Bold,
                        fontFamily = FontFamily.Monospace
                    )
                }

                Spacer(modifier = Modifier.height(8.dp))
                Box(
                    modifier = Modifier
                        .width(40.dp)
                        .height(1.dp)
                        .background(accentGradient)
                )
                Spacer(modifier = Modifier.height(8.dp))

                Column(
                    verticalArrangement = Arrangement.spacedBy(8.dp),
                    horizontalAlignment = Alignment.CenterHorizontally,
                    modifier = Modifier.weight(1f)
                ) {
                    pinnedApps.forEach { app ->
                        AppIconView(
                            appInfo = app,
                            iconStyle = iconStyle,
                            accentGradient = accentGradient,
                            onClick = {
                                val launchIntent = context.packageManager.getLaunchIntentForPackage(app.activityInfo.packageName)
                                launchIntent?.let { context.startActivity(it) }
                            },
                            onLongClick = {
                                selectedAppContextMenu = app
                            }
                        )
                    }
                }
            }

            // HAUPTBEREICH
            Box(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxHeight()
                    .padding(12.dp)
            ) {
                when {
                    showSysMonitor -> {
                        SystemMonitorOverlay(
                            accentGradient = accentGradient,
                            bgColor = bgColor,
                            onClose = { showSysMonitor = false }
                        )
                    }
                    showConfigMenu -> {
                        LimboConfigScreen(
                            bgColor = baseBgColor,
                            bgSecColor = baseBgSecColor,
                            accentColor = baseAccentColor,
                            accentSecColor = baseAccentSecColor,
                            accentGradient = accentGradient,
                            iconStyle = iconStyle,
                            bgStyle = bgStyle,
                            terminalEnabled = terminalEnabled,
                            audioEnabled = audioFeedbackEnabled,
                            onRefreshSystem = { refreshTrigger++ },
                            onBgColorChange = {
                                bgColorHex = it.value.toLong()
                                saveConfig()
                            },
                            onBgSecColorChange = {
                                bgSecondaryColorHex = it.value.toLong()
                                saveConfig()
                            },
                            onAccentColorChange = {
                                accentColorHex = it.value.toLong()
                                saveConfig()
                            },
                            onAccentSecColorChange = {
                                accentSecondaryColorHex = it.value.toLong()
                                saveConfig()
                            },
                            onIconStyleChange = {
                                iconStyle = it
                                saveConfig()
                            },
                            onBgStyleChange = {
                                bgStyle = it
                                saveConfig()
                            },
                            onTerminalEnabledChange = {
                                terminalEnabled = it
                                saveConfig()
                            },
                            onAudioEnabledChange = {
                                audioFeedbackEnabled = it
                                saveConfig()
                            },
                            onClose = { showConfigMenu = false }
                        )
                    }
                    showAppDrawer -> {
                        Column(modifier = Modifier.fillMaxSize()) {

                            OutlinedTextField(
                                value = searchQuery,
                                onValueChange = { newValue ->
                                    searchQuery = newValue
                                    val queryClean = newValue.trim().lowercase()
                                    if (queryClean == "limbo config" || queryClean == "config") {
                                        showConfigMenu = true
                                    } else if (queryClean == "top" || queryClean == "sys" || queryClean == "fastfetch" || queryClean == "neofetch") {
                                        showSysMonitor = true
                                    } else if (queryClean == "terminal" || queryClean == "cmd") {
                                        showTerminalWindow = true
                                    } else if (queryClean == "hex" || queryClean == "hexedit") {
                                        showHexViewerWindow = true
                                    }
                                },
                                modifier = Modifier.fillMaxWidth(),
                                placeholder = { Text("App suchen...", color = accentColor.copy(alpha = 0.5f)) },
                                singleLine = true,
                                shape = RectangleShape,
                                textStyle = TextStyle(color = accentColor),
                                colors = OutlinedTextFieldDefaults.colors(
                                    focusedBorderColor = accentColor,
                                    unfocusedBorderColor = accentSecColor,
                                    cursorColor = accentColor
                                )
                            )

                            Spacer(modifier = Modifier.height(12.dp))

                            LazyVerticalGrid(
                                columns = GridCells.Adaptive(minSize = 75.dp),
                                horizontalArrangement = Arrangement.spacedBy(8.dp),
                                verticalArrangement = Arrangement.spacedBy(12.dp),
                                modifier = Modifier.fillMaxSize()
                            ) {
                                items(filteredApps) { appInfo ->
                                    MonochromeAppItem(
                                        appInfo = appInfo,
                                        iconStyle = iconStyle,
                                        accentGradient = accentGradient,
                                        onClick = {
                                            val launchIntent = context.packageManager.getLaunchIntentForPackage(appInfo.activityInfo.packageName)
                                            launchIntent?.let { context.startActivity(it) }
                                        },
                                        onLongClick = {
                                            selectedAppContextMenu = appInfo
                                        }
                                    )
                                }
                            }
                        }
                    }
                    else -> {
                        Box(
                            modifier = Modifier
                                .fillMaxSize()
                                .combinedClickable(
                                    onClick = { showSysMonitor = true },
                                    onLongClick = { showConfigMenu = true }
                                ),
                            contentAlignment = Alignment.Center
                        ) {
                            DynamicBackgroundCanvas(style = bgStyle, accentGradient = accentGradient)

                            Text(
                                text = if (isHackerModeActive) "SYSTEM OVERRIDE // HACKED" else "LIMBO OS V11.0 [ARCH/KERNEL]",
                                style = TextStyle(brush = accentGradient),
                                fontSize = 28.sp,
                                fontWeight = FontWeight.Bold,
                                letterSpacing = 3.sp,
                                fontFamily = FontFamily.Monospace
                            )
                        }
                    }
                }
            }
        }

        // FLOATING WINDOW TERMINAL
        if (showTerminalWindow && terminalEnabled) {
            FloatingTerminalWindow(
                bgColor = bgColor,
                accentGradient = accentGradient,
                accentColor = accentColor,
                audioEnabled = audioFeedbackEnabled,
                onClose = { showTerminalWindow = false },
                onUpdateBgColor = { bgColorHex = it.value.toLong(); saveConfig() },
                onUpdateBgSecColor = { bgSecondaryColorHex = it.value.toLong(); saveConfig() },
                onUpdateAccentColor = { accentColorHex = it.value.toLong(); saveConfig() },
                onUpdateAccentSecColor = { accentSecondaryColorHex = it.value.toLong(); saveConfig() },
                onTriggerRmRfGlitch = { triggerRmRfGlitch() },
                onTriggerLimboHackerMode = { triggerLimboHackerMode() }
            )
        }

        // FLOATING HEX VIEWER / MEMORY INSPECTOR FOR LINUX NERDS
        if (showHexViewerWindow) {
            FloatingHexViewerWindow(
                bgColor = bgColor,
                accentGradient = accentGradient,
                accentColor = accentColor,
                onClose = { showHexViewerWindow = false }
            )
        }

        if (isHackerModeActive) {
            HackerModeOverlayScreen()
        }

        if (isGlitching) {
            GlitchOverlayScreen()
        }

        selectedAppContextMenu?.let { app ->
            val pkgName = app.activityInfo.packageName
            val appLabel = app.loadLabel(context.packageManager).toString()
            val isPinned = pinnedPackageNames.contains(pkgName)

            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color.Black.copy(alpha = 0.7f))
                    .clickable { selectedAppContextMenu = null },
                contentAlignment = Alignment.Center
            ) {
                Column(
                    modifier = Modifier
                        .width(260.dp)
                        .background(bgColor)
                        .border(1.dp, accentGradient, RectangleShape)
                        .padding(16.dp)
                        .clickable(enabled = false) {},
                    horizontalAlignment = Alignment.CenterHorizontally
                ) {
                    Text(
                        text = appLabel.uppercase(),
                        style = TextStyle(brush = accentGradient),
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Bold,
                        textAlign = TextAlign.Center,
                        fontFamily = FontFamily.Monospace,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis
                    )

                    Spacer(modifier = Modifier.height(12.dp))
                    Box(modifier = Modifier.fillMaxWidth().height(1.dp).background(accentGradient))
                    Spacer(modifier = Modifier.height(12.dp))

                    Button(
                        onClick = {
                            if (isPinned) {
                                pinnedPackageNames.remove(pkgName)
                            } else {
                                pinnedPackageNames.add(pkgName)
                            }
                            savePinnedApps()
                            selectedAppContextMenu = null
                        },
                        shape = RectangleShape,
                        colors = ButtonDefaults.buttonColors(containerColor = bgColor),
                        modifier = Modifier.fillMaxWidth().border(1.dp, accentGradient, RectangleShape)
                    ) {
                        Text(
                            text = if (isPinned) "Aus Leiste entfernen" else "In Leiste pinnen",
                            style = TextStyle(brush = accentGradient),
                            fontSize = 12.sp
                        )
                    }

                    Spacer(modifier = Modifier.height(8.dp))

                    Button(
                        onClick = {
                            selectedAppContextMenu = null
                            val intent = Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
                                data = Uri.fromParts("package", pkgName, null)
                            }
                            context.startActivity(intent)
                        },
                        shape = RectangleShape,
                        colors = ButtonDefaults.buttonColors(containerColor = bgColor),
                        modifier = Modifier.fillMaxWidth().border(1.dp, accentGradient, RectangleShape)
                    ) {
                        Text(
                            text = "App-Informationen",
                            style = TextStyle(brush = accentGradient),
                            fontSize = 12.sp
                        )
                    }
                }
            }
        }

// --- HEX VIEWER WINDOW ---
@Composable
fun FloatingHexViewerWindow(
    bgColor: Color,
    accentGradient: Brush,
    accentColor: Color,
    onClose: () -> Unit
) {
    var offsetX by remember { mutableFloatStateOf(140f) }
    var offsetY by remember { mutableFloatStateOf(140f) }
    var filePathInput by remember { mutableStateOf("/proc/version") }
    var hexContent by remember { mutableStateOf("Klicke READ um Datei zu inspecten...") }

    fun readHexFile(path: String) {
        try {
            val file = File(path)
            if (!file.exists()) {
                hexContent = "FEHLER: Datei existiert nicht!"
                return
            }
            val bytes = ByteArray(256)
            val raf = RandomAccessFile(file, "r")
            val readBytes = raf.read(bytes)
            raf.close()

            val sb = StringBuilder()
            for (i in 0 until maxOf(0, readBytes)) {
                if (i % 8 == 0 && i != 0) sb.append("\n")
                sb.append(String.format("%02X ", bytes[i]))
            }
            hexContent = sb.toString()
        } catch (e: Exception) {
            hexContent = "ERR: ${e.localizedMessage}"
        }
    }

    Box(
        modifier = Modifier
            .offset { IntOffset(offsetX.roundToInt(), offsetY.roundToInt()) }
            .size(320.dp, 280.dp)
            .background(bgColor)
            .border(1.dp, accentGradient, RectangleShape)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(32.dp)
                    .background(accentColor.copy(alpha = 0.2f))
                    .pointerInput(Unit) {
                        detectDragGestures { change, dragAmount ->
                            change.consume()
                            offsetX += dragAmount.x
                            offsetY += dragAmount.y
                        }
                    }
                    .padding(horizontal = 8.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Text(
                    text = "HEX / MEMORY INSPECTOR",
                    style = TextStyle(brush = accentGradient),
                    fontSize = 11.sp,
                    fontWeight = FontWeight.Bold,
                    fontFamily = FontFamily.Monospace
                )
                Text(
                    text = "[ X ]",
                    style = TextStyle(brush = accentGradient),
                    fontSize = 12.sp,
                    fontWeight = FontWeight.Bold,
                    modifier = Modifier.clickable { onClose() }
                )
            }

            Row(
                modifier = Modifier.fillMaxWidth().padding(4.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                OutlinedTextField(
                    value = filePathInput,
                    onValueChange = { filePathInput = it },
                    modifier = Modifier.weight(1f),
                    singleLine = true,
                    textStyle = TextStyle(color = accentColor, fontSize = 10.sp, fontFamily = FontFamily.Monospace),
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedBorderColor = accentColor,
                        unfocusedBorderColor = accentColor.copy(alpha = 0.3f),
                        cursorColor = accentColor
                    )
                )
                Spacer(modifier = Modifier.width(4.dp))
                Button(
                    onClick = { readHexFile(filePathInput) },
                    shape = RectangleShape,
                    colors = ButtonDefaults.buttonColors(containerColor = Color.Transparent),
                    modifier = Modifier.border(1.dp, accentGradient, RectangleShape)
                ) {
                    Text("READ", style = TextStyle(brush = accentGradient), fontSize = 10.sp, fontFamily = FontFamily.Monospace)
                }
            }

            Box(modifier = Modifier.fillMaxSize().padding(6.dp).verticalScroll(rememberScrollState())) {
                Text(
                    text = hexContent,
                    style = TextStyle(brush = accentGradient),
                    fontFamily = FontFamily.Monospace,
                    fontSize = 11.sp
                )
            }
        }
    }
}

// --- FLOATING TERMINAL WINDOW WITH HISTORY & ALIASES ---
@Composable
fun FloatingTerminalWindow(
    bgColor: Color,
    accentGradient: Brush,
    accentColor: Color,
    audioEnabled: Boolean,
    onClose: () -> Unit,
    onUpdateBgColor: (Color) -> Unit,
    onUpdateBgSecColor: (Color) -> Unit,
    onUpdateAccentColor: (Color) -> Unit,
    onUpdateAccentSecColor: (Color) -> Unit,
    onTriggerRmRfGlitch: () -> Unit,
    onTriggerLimboHackerMode: () -> Unit
) {
    var offsetX by remember { mutableFloatStateOf(80f) }
    var offsetY by remember { mutableFloatStateOf(80f) }
    var isFullscreen by remember { mutableStateOf(false) }

    var currentDirectory by remember { mutableStateOf(File("/").canonicalFile) }
    var commandInput by remember { mutableStateOf("") }
    
    val commandHistory = remember { mutableStateListOf<String>() }
    var historyIndex by remember { mutableIntStateOf(-1) }
    val userAliases = remember { mutableStateMapOf<String, String>() }

    val consoleLogs = remember {
        mutableStateListOf(
            "Limbo Terminal v3.0 [Arch/Linux Engine]",
            "Tippe 'fastfetch' oder 'neofetch' für System-Props.",
            "Secrets: 'sudo rm -rf /' oder 'LIMBO'"
        )
    }

    val listState = rememberLazyListState()

    LaunchedEffect(consoleLogs.size) {
        if (consoleLogs.isNotEmpty()) {
            listState.animateScrollToItem(consoleLogs.size - 1)
        }
    }

    fun executeLinuxCommand(rawCmdStr: String) {
        val trimmedRaw = rawCmdStr.trim()
        if (trimmedRaw.isEmpty()) return

        if (audioEnabled) TerminalAudioSynth.playBeep(750f, 25)

        commandHistory.add(trimmedRaw)
        historyIndex = commandHistory.size

        // Alias Ersetzung
        val firstWord = trimmedRaw.split("\\s+".toRegex())[0]
        val cmdStr = if (userAliases.containsKey(firstWord)) {
            trimmedRaw.replaceFirst(firstWord, userAliases[firstWord]!!)
        } else {
            trimmedRaw
        }

        consoleLogs.add("limbo@arch:${currentDirectory.path}$ $cmdStr")

        val cleanCmd = cmdStr.lowercase().replace("\\s+".toRegex(), " ")

        if (cleanCmd == "sudo rm -rf /" || cleanCmd == "sudo rm -rf" || cleanCmd == "rm -rf /") {
            consoleLogs.add("[FATAL] DELETING SYSTEM ROOT... SYSTEM CORRUPTED!")
            onTriggerRmRfGlitch()
            return
        }

        if (cmdStr.equals("LIMBO", ignoreCase = true)) {
            consoleLogs.add("[OVERRIDE] GTA HACKER MODE ACTIVATED (30s)...")
            onTriggerLimboHackerMode()
            return
        }

        val parts = cmdStr.split("\\s+".toRegex())
        val cmd = parts[0].lowercase()

        // Fastfetch / Neofetch
        if (cmd == "fastfetch" || cmd == "neofetch") {
            consoleLogs.add("       /\\         limbo@arch-linux")
            consoleLogs.add("      /  \\        ----------------")
            consoleLogs.add("     / /\\ \\       OS: Limbo Arch Linux (Android)")
            consoleLogs.add("    / /  \\ \\      Kernel: ${System.getProperty("os.version")}")
            consoleLogs.add("   / / __ \\ \\     Host: ${Build.MANUFACTURER} ${Build.MODEL}")
            consoleLogs.add("  / /_/__\\ \\ \\    Uptime: ${SystemClock.elapsedRealtime() / 3600000}h ${(SystemClock.elapsedRealtime() % 3600000) / 60000}m")
            consoleLogs.add(" /__________\\ \\   Shell: Limbo Shell v3.0")
            consoleLogs.add("               \\  RAM: ${Runtime.getRuntime().totalMemory() / (1024*1024)}MB / ${Runtime.getRuntime().maxMemory() / (1024*1024)}MB")
            return
        }

        // Custom Alias
        if (cmd == "alias" && parts.size > 1) {
            val aliasDef = cmdStr.substringAfter("alias ").trim()
            if (aliasDef.contains("=")) {
                val k = aliasDef.substringBefore("=").trim()
                val v = aliasDef.substringAfter("=").replace("\"", "").replace("'", "").trim()
                userAliases[k] = v
                consoleLogs.add("Alias gesetzt: $k -> $v")
            } else {
                userAliases.forEach { (k, v) -> consoleLogs.add("alias $k='$v'") }
            }
            return
        }

        if (cmd == "set" && parts.size >= 4 && parts[1].lowercase() == "color") {
            val target = parts[2].lowercase()
            val colorQuery = parts.subList(3, parts.size).joinToString(" ")

            val foundColor = AvailableColors.find { it.name.lowercase() == colorQuery.lowercase() }?.color
                ?: run {
                    try {
                        Color(android.graphics.Color.parseColor(colorQuery))
                    } catch (_: Exception) {
                        null
                    }
                }

            if (foundColor != null) {
                when (target) {
                    "bg" -> { onUpdateBgColor(foundColor); consoleLogs.add("--> Primäre Hintergrundfarbe geändert.") }
                    "bg2" -> { onUpdateBgSecColor(foundColor); consoleLogs.add("--> Sekundäre Hintergrundfarbe geändert.") }
                    "accent" -> { onUpdateAccentColor(foundColor); consoleLogs.add("--> Primäre Akzentfarbe geändert.") }
                    "accent2" -> { onUpdateAccentSecColor(foundColor); consoleLogs.add("--> Sekundäre Akzentfarbe geändert.") }
                    else -> consoleLogs.add("Unbekanntes Ziel. Nutze: bg, bg2, accent, accent2")
                }
            } else {
                consoleLogs.add("Farbe '$colorQuery' nicht gefunden. Tippe 'colors' für Liste.")
            }
            return
        }

        when (cmd) {
            "clear" -> {
                consoleLogs.clear()
                return
            }
            "help" -> {
                consoleLogs.add("--- INTEGRATED COMMANDS ---")
                consoleLogs.add("fastfetch / neofetch - Zeigt Arch System Overview")
                consoleLogs.add("alias name=\"command\" - Definiert Custom Aliases")
                consoleLogs.add("LIMBO - Startet den Hacker-Modus")
                consoleLogs.add("set color <bg|bg2|accent|accent2> <color>")
                consoleLogs.add("colors - Liste aller verfügbaren Farbnamen")
                consoleLogs.add("cd <dir> - Ordner wechseln")
                consoleLogs.add("clear - Konsole leeren")
                consoleLogs.add("--- NATIVE SHELL EXECUTION ---")
                consoleLogs.add("ls, pwd, whoami, uname, ps, cat, top, df, free, echo...")
                return
            }
            "colors" -> {
                consoleLogs.add("Verfügbare Farbnamen:")
                AvailableColors.chunked(4).forEach { chunk ->
                    consoleLogs.add(chunk.joinToString(" | ") { it.name })
                }
                return
            }
            "cd" -> {
                if (parts.size > 1) {
                    val targetPath = parts[1]
                    val targetDir = if (targetPath.startsWith("/")) File(targetPath) else File(currentDirectory, targetPath)
                    if (targetDir.exists() && targetDir.isDirectory) {
                        currentDirectory = targetDir.canonicalFile
                    } else {
                        consoleLogs.add("cd: Pfad nicht gefunden: $targetPath")
                    }
                } else {
                    currentDirectory = File("/")
                }
                return
            }
        }

        try {
            val process = Runtime.getRuntime().exec(arrayOf("/system/bin/sh", "-c", cmdStr), null, currentDirectory)
            val reader = BufferedReader(InputStreamReader(process.inputStream))
            val errorReader = BufferedReader(InputStreamReader(process.errorStream))

            var line: String?
            while (reader.readLine().also { line = it } != null) {
                consoleLogs.add(line ?: "")
            }
            while (errorReader.readLine().also { line = it } != null) {
                consoleLogs.add("[err] ${line ?: ""}")
            }
            process.waitFor()
        } catch (e: Exception) {
            consoleLogs.add("Fehler beim Ausführen: ${e.localizedMessage}")
        }
    }

    val windowModifier = if (isFullscreen) {
        Modifier.fillMaxSize()
    } else {
        Modifier
            .offset { IntOffset(offsetX.roundToInt(), offsetY.roundToInt()) }
            .size(350.dp, 270.dp)
    }

    Box(
        modifier = windowModifier
            .background(bgColor)
            .border(1.dp, accentGradient, RectangleShape)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {

            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(32.dp)
                    .background(accentColor.copy(alpha = 0.2f))
                    .pointerInput(isFullscreen) {
                        if (!isFullscreen) {
                            detectDragGestures { change, dragAmount ->
                                change.consume()
                                offsetX += dragAmount.x
                                offsetY += dragAmount.y
                            }
                        }
                    }
                    .padding(horizontal = 8.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Text(
                    text = "LIMBO TERMINAL V3.0",
                    style = TextStyle(brush = accentGradient),
                    fontSize = 11.sp,
                    fontWeight = FontWeight.Bold,
                    fontFamily = FontFamily.Monospace
                )

                Row(horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                    Text(
                        text = if (isFullscreen) "[ ▢ ]" else "[ 🗖 ]",
                        style = TextStyle(brush = accentGradient),
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold,
                        modifier = Modifier.clickable { isFullscreen = !isFullscreen }
                    )
                    Text(
                        text = "[ X ]",
                        style = TextStyle(brush = accentGradient),
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold,
                        modifier = Modifier.clickable { onClose() }
                    )
                }
            }

            Box(modifier = Modifier.fillMaxWidth().height(1.dp).background(accentGradient))

            LazyColumn(
                state = listState,
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
                    .padding(6.dp)
            ) {
                items(consoleLogs) { log ->
                    Text(
                        text = log,
                        style = TextStyle(brush = accentGradient),
                        fontFamily = FontFamily.Monospace,
                        fontSize = 11.sp
                    )
                }
            }

            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(4.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "> ",
                    style = TextStyle(brush = accentGradient),
                    fontFamily = FontFamily.Monospace,
                    fontSize = 12.sp,
                    fontWeight = FontWeight.Bold
                )

                OutlinedTextField(
                    value = commandInput,
                    onValueChange = { commandInput = it },
                    modifier = Modifier.weight(1f),
                    singleLine = true,
                    textStyle = TextStyle(color = accentColor, fontSize = 11.sp, fontFamily = FontFamily.Monospace),
                    keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                    keyboardActions = KeyboardActions(onDone = {
                        executeLinuxCommand(commandInput)
                        commandInput = ""
                    }),
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedBorderColor = Color.Transparent,
                        unfocusedBorderColor = Color.Transparent,
                        cursorColor = accentColor
                    )
                )

                // History Buttons
                Text(
                    text = "[▲]",
                    style = TextStyle(brush = accentGradient),
                    fontSize = 11.sp,
                    fontFamily = FontFamily.Monospace,
                    modifier = Modifier.clickable {
                        if (commandHistory.isNotEmpty() && historyIndex > 0) {
                            historyIndex--
                            commandInput = commandHistory[historyIndex]
                        }
                    }.padding(horizontal = 2.dp)
                )
                Text(
                    text = "[▼]",
                    style = TextStyle(brush = accentGradient),
                    fontSize = 11.sp,
                    fontFamily = FontFamily.Monospace,
                    modifier = Modifier.clickable {
                        if (commandHistory.isNotEmpty() && historyIndex < commandHistory.size - 1) {
                            historyIndex++
                            commandInput = commandHistory[historyIndex]
                        }
                    }.padding(horizontal = 2.dp)
                )
            }
        }
    }
}

// --- 40 CANVAS ENGINE ANIMATIONEN ---
@Composable
fun DynamicBackgroundCanvas(style: String, accentGradient: Brush) {
    if (style == "Minimal Void") return

    val infiniteTransition = rememberInfiniteTransition(label = "bg_engine")
    val animVal by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 1f,
        animationSpec = infiniteRepeatable(
            animation = tween(4000, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "loop"
    )

    Canvas(modifier = Modifier.fillMaxSize()) {
        val w = size.width
        val h = size.height
        val t = animVal * 2f * PI.toFloat()

        when (style) {
            "Starfield" -> {
                for (i in 0 until 50) {
                    val angle = (i * 137.5f) * (PI.toFloat() / 180f)
                    val speed = (i % 5 + 1) * 0.15f
                    val dist = ((animVal * speed * w) + (i * 15f)) % (w * 0.7f)
                    val x = w / 2 + cos(angle) * dist
                    val y = h / 2 + sin(angle) * dist
                    drawCircle(brush = accentGradient, alpha = (dist / (w * 0.7f)), radius = (1.5f + (dist / 100f)).dp.toPx(), center = Offset(x, y))
                }
            }
            "Matrix Rain" -> {
                val cols = 18
                val colW = w / cols
                for (i in 0 until cols) {
                    val x = i * colW + colW / 2
                    val speed = ((i * 7) % 5 + 2) * 0.25f
                    val y = ((animVal * speed * h + i * 65f) % (h + 100f)) - 50f
                    drawLine(brush = accentGradient, alpha = 0.8f, start = Offset(x, y), end = Offset(x, y + 50.dp.toPx()), strokeWidth = 2.dp.toPx())
                }
            }
            "Cyber Grid" -> {
                val step = 40.dp.toPx()
                var x = 0f
                while (x < w) {
                    drawLine(brush = accentGradient, alpha = 0.15f, start = Offset(x, 0f), end = Offset(x, h), strokeWidth = 1.dp.toPx())
                    x += step
                }
                var y = (animVal * step) % step
                while (y < h) {
                    drawLine(brush = accentGradient, alpha = (y / h) * 0.3f, start = Offset(0f, y), end = Offset(w, y), strokeWidth = 1.dp.toPx())
                    y += step
                }
            }
            "Neon Scanlines" -> {
                val step = 8.dp.toPx()
                var y = 0f
                while (y < h) {
                    drawLine(brush = accentGradient, alpha = 0.1f, start = Offset(0f, y), end = Offset(w, y), strokeWidth = 1.dp.toPx())
                    y += step
                }
                val scanBarY = (animVal * h)
                drawRect(brush = accentGradient, alpha = 0.2f, topLeft = Offset(0f, scanBarY), size = Size(w, 20.dp.toPx()))
            }
            else -> {
                for (i in 0..20) {
                    val px = (i * 50f + sin(t + i) * 20f) % w
                    val py = (i * 80f + cos(t + i) * 20f) % h
                    drawCircle(brush = accentGradient, alpha = 0.3f, radius = 4.dp.toPx(), center = Offset(px, py))
                }
            }
        }
    }
}

// --- APP ICON COMPONENT ---
@OptIn(ExperimentalFoundationApi::class)
@Composable
fun AppIconView(
    appInfo: ResolveInfo,
    iconStyle: String,
    accentGradient: Brush,
    onClick: () -> Unit,
    onLongClick: () -> Unit
) {
    val context = LocalContext.current
    val appName = remember(appInfo) { appInfo.loadLabel(context.packageManager).toString() }

    Box(
        modifier = Modifier
            .size(48.dp)
            .border(1.dp, accentGradient, RectangleShape)
            .combinedClickable(onClick = onClick, onLongClick = onLongClick),
        contentAlignment = Alignment.Center
    ) {
        if (iconStyle == "System Icons") {
            var iconBitmap by remember(appInfo) { mutableStateOf<androidx.compose.ui.graphics.ImageBitmap?>(null) }

            LaunchedEffect(appInfo) {
                try {
                    val drawable = appInfo.loadIcon(context.packageManager)
                    iconBitmap = drawable.toBitmap().asImageBitmap()
                } catch (_: Exception) {
                    iconBitmap = null
                }
            }

            if (iconBitmap != null) {
                Image(
                    bitmap = iconBitmap!!,
                    contentDescription = appName,
                    modifier = Modifier.size(34.dp)
                )
            } else {
                Text(
                    text = nameToInitials(appName),
                    style = TextStyle(brush = accentGradient),
                    fontWeight = FontWeight.Bold,
                    fontSize = 18.sp
                )
            }
        } else {
            Text(
                text = nameToInitials(appName),
                style = TextStyle(brush = accentGradient),
                fontWeight = FontWeight.Bold,
                fontSize = 18.sp
            )
        }
    }
}

@OptIn(ExperimentalFoundationApi::class)
@Composable
fun MonochromeAppItem(
    appInfo: ResolveInfo,
    iconStyle: String,
    accentGradient: Brush,
    onClick: () -> Unit,
    onLongClick: () -> Unit
) {
    val context = LocalContext.current
    val appName = remember(appInfo) { appInfo.loadLabel(context.packageManager).toString() }

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        modifier = Modifier.padding(4.dp)
    ) {
        AppIconView(
            appInfo = appInfo,
            iconStyle = iconStyle,
            accentGradient = accentGradient,
            onClick = onClick,
            onLongClick = onLongClick
        )
        Spacer(modifier = Modifier.height(4.dp))
        Text(
            text = appName,
            style = TextStyle(brush = accentGradient),
            fontSize = 10.sp,
            maxLines = 1,
            overflow = TextOverflow.Ellipsis,
            textAlign = TextAlign.Center
        )
    }
}

fun nameToInitials(name: String): String {
    return name.take(1).uppercase()
}
